    <script>
        // --- GOOGLE SHEETS INTEGRATION URL ---
        // ⚠️ DİKKAT: BURAYA GOOGLE APPS SCRIPT WEB APP URL'NİZİ YAPIŞTIRIN! ⚠️
        const GOOGLE_APP_SCRIPT_URL = 'YOUR_GOOGLE_SCRIPT_WEB_APP_URL_HERE';
        // -------------------------------------

        document.addEventListener('DOMContentLoaded', () => {
            const form = document.getElementById('subscribeForm');
            const emailInput = document.getElementById('emailInput');
            const formContainer = document.getElementById('formContainer');
            const successContainer = document.getElementById('successContainer');
            const registeredEmailSpan = document.getElementById('registeredEmail');
            const resetFormBtn = document.getElementById('resetFormBtn');

            form.addEventListener('submit', (e) => {
                e.preventDefault(); 
                
                const email = emailInput.value.trim();
                
                if(email) {
                    const submitBtn = form.querySelector('button[type="submit"]');
                    const originalBtnContent = submitBtn.innerHTML;
                    
                    submitBtn.innerHTML = '<i class="fa-solid fa-circle-notch fa-spin"></i> Processing...';
                    submitBtn.disabled = true;

                    // Send data to Google Sheets API
                    fetch(GOOGLE_APP_SCRIPT_URL, {
                        method: 'POST',
                        mode: 'no-cors', 
                        cache: 'no-cache',
                        headers: {
                            'Content-Type': 'application/json'
                        },
                        redirect: 'follow',
                        body: JSON.stringify({ email: email })
                    })
                    .then(() => {
                        registeredEmailSpan.textContent = email;
                        
                        formContainer.classList.add('opacity-0', 'scale-95');
                        
                        setTimeout(() => {
                            formContainer.classList.add('hidden');
                            formContainer.classList.remove('opacity-0', 'scale-95');
                            
                            successContainer.classList.remove('hidden');
                            setTimeout(() => {
                                successContainer.classList.add('opacity-100', 'scale-100');
                            }, 50);
                        }, 300);

                        submitBtn.innerHTML = originalBtnContent;
                        submitBtn.disabled = false;
                        emailInput.value = '';
                    })
                    .catch((error) => {
                        console.error('Error:', error);
                        submitBtn.innerHTML = originalBtnContent;
                        submitBtn.disabled = false;
                        alert("An error occurred while subscribing. Please try again later.");
                    });
                }
            });

            resetFormBtn.addEventListener('click', () => {
                successContainer.classList.remove('opacity-100', 'scale-100');
                successContainer.classList.add('opacity-0', 'scale-95');
                
                setTimeout(() => {
                    successContainer.classList.add('hidden');
                    successContainer.classList.remove('opacity-0', 'scale-95');
                    
                    formContainer.classList.remove('hidden');
                    setTimeout(() => {
                        formContainer.classList.add('opacity-100', 'scale-100');
                        emailInput.focus();
                    }, 50);
                }, 300);
            });
            
            // Navbar scroll effect
            const navbar = document.getElementById('navbar');
            window.addEventListener('scroll', () => {
                if (window.scrollY > 20) {
                    navbar.classList.add('shadow-lg', 'bg-slate-900/90');
                    navbar.classList.remove('glass');
                } else {
                    navbar.classList.remove('shadow-lg', 'bg-slate-900/90');
                    navbar.classList.add('glass');
                }
            });
        });
    </script>
